# Esta es ahora la misma configuracion que SUBIR-A-HOSTALIA/.htaccess: se
# mantiene aqui como copia de referencia. Si alguna vez tienes que
# regenerarla a mano, recuerda que el hash sha256 de script-src debe
# recalcularse cada vez que se edite el <script> de index.html o firmar.html,
# o el navegador bloqueara la pagina entera (pantalla en blanco).
# NOTA: se han eliminado a proposito RequestReadTimeout, Options y
# ServerSignature porque muchos alojamientos compartidos los prohiben en
# .htaccess y devuelven error 500.

DirectoryIndex index.html

# --- Forzar HTTPS ----------------------------------------------------------
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteCond %{HTTP:X-Forwarded-SSL} !on
  RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]
</IfModule>

# --- Cabeceras de seguridad ------------------------------------------------
<IfModule mod_headers.c>
  Header always set Content-Security-Policy "default-src 'none'; script-src 'self' 'sha256-9/P1lSh4bLQEL59e0xwq+heosNsCACMrwXNFA8W3308=' 'sha256-FClgaDpupbLx18rPepL1KLcQUnwWxYhIB4wLBL7DWfA='; style-src 'self' 'unsafe-inline'; img-src 'self' data:; font-src 'self'; connect-src 'self' https://ezwhkpfqnzyfulivlows.supabase.co; form-action 'none'; frame-ancestors 'none'; base-uri 'none'; object-src 'none'"
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "DENY"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Cross-Origin-Opener-Policy "same-origin"
  Header always set Cross-Origin-Resource-Policy "same-origin"
  Header always set Permissions-Policy "geolocation=(), camera=(), microphone=(), payment=(), usb=()"
  Header always set X-Permitted-Cross-Domain-Policies "none"

# --- Bloqueo de ficheros sensibles (sintaxis doble 2.2 / 2.4) --------------
<FilesMatch "(^\.|\.(md|zip|log|bak|sql|env|ini|yml|py|sh)$)">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# --- Compresion ------------------------------------------------------------
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript
</IfModule>
